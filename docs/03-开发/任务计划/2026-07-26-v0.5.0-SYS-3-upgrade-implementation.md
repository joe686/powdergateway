# v0.5.0 · SYS-3 权限体系升级 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 SYS-3 从"三固定角色 + 硬编码菜单白名单"升级为"角色/菜单/授权关系全 DB 驱动，含菜单分级、授权继承、分级授权"。

**Architecture:** 新建 4 张表（sys_menu / sys_role / sys_user_role / sys_role_menu）统一菜单元模型。授权继承走"存父不存子，查询时递归展开"。分级授权走 4 层校验。一次性数据迁移把 MenuPermission.java 常量表 + testkit 硬编码菜单迁入 DB，迁移前后可见菜单集合逐条断言相等，零权限漂移。

**Tech Stack:** Spring Boot 2.7 / MyBatis-Plus / Sa-Token / H2(test) / Vue 3 / Element Plus / Pinia

```java
@SpringBootTest @ActiveProfiles("test") @Transactional
class SysMenuServiceTest {
    @Autowired SysMenuService svc;
    @Autowired SysMenuMapper mapper;

    @Test void 保存_返回_id() {
        SysMenu m = new SysMenu(); m.setCode("t.a"); m.setName("A"); m.setType("PAGE"); m.setOrigin("NATIVE");
        Long id = svc.save(m);
        assertThat(id).isNotNull();
    }

    @Test void 删除内置菜单_抛异常() {
        SysMenu m = new SysMenu(); m.setCode("b.core"); m.setName("Core"); m.setType("PAGE"); m.setOrigin("NATIVE"); m.setIsBuiltin(1);
        Long id = svc.save(m);
        assertThatThrownBy(() -> svc.delete(id))
            .isInstanceOf(BusinessException.class)
            .hasMessageContaining("内置菜单不可删");
    }

    @Test void 递归展开子菜单() {
        // 构造 root → gr → item 三层
        Long root = svc.save(newMenu("t.root", null));
        Long gr = svc.save(newMenu("t.root.gr", root));
        Long item = svc.save(newMenu("t.root.gr.item", gr));
```

```java
@Service @RequiredArgsConstructor
public class AuthMenuService {
    private final SysUserRoleMapper userRoleMapper;
    private final SysRoleMenuMapper roleMenuMapper;
    private final SysMenuMapper menuMapper;
    private final SysMenuService menuService;

    public List<MenuTreeNode> getMenuTreeForUser(Long userId) {
        List<Long> roleIds = userRoleMapper.selectRoleIdsByUserId(userId);
        if (roleIds.isEmpty()) return Collections.emptyList();

        List<Long> explicitMenuIds = roleMenuMapper.selectMenuIdsByRoleIds(roleIds);
        Set<Long> allMenuIds = menuService.expandChildren(explicitMenuIds);
        allMenuIds.addAll(explicitMenuIds);

        List<SysMenu> menus = menuMapper.selectByIds(allMenuIds).stream()
            .filter(m -> m.getVisible() == 1)
            .filter(m -> !"DISABLED".equals(m.getStatus()) && !"ORPHAN".equals(m.getStatus()))
            .collect(Collectors.toList());

        return buildTree(menus);
    }

    private List<MenuTreeNode> buildTree(List<SysMenu> menus) {
        Map<Long, MenuTreeNode> nodes = menus.stream()
            .collect(Collectors.toMap(SysMenu::getId, this::toNode));
        List<MenuTreeNode> roots = new ArrayList<>();
        for (SysMenu m : menus) {
            MenuTreeNode node = nodes.get(m.getId());
            if (m.getParentId() == null) roots.add(node);
            else Optional.ofNullable(nodes.get(m.getParentId())).ifPresent(p -> p.getChildren().add(node));
        }
        return roots;
    }

    private MenuTreeNode toNode(SysMenu m) {
        MenuTreeNode n = new MenuTreeNode();
        BeanUtils.copyProperties(m, n);
        return n;
    }
}
```

```java
@Transactional
public void grantMenus(Long currentUserId, Long targetRoleId, List<Long> menuIds) {
    // 层 1: can_grant 校验
    boolean hasGrant = userRoleMapper.selectRoleIdsByUserId(currentUserId).stream()
        .map(roleMapper::selectById)
        .anyMatch(r -> Integer.valueOf(1).equals(r.getCanGrant()));
    if (!hasGrant) throw new BusinessException(403, "无授权能力");

    // 层 2: 拥有集展开
    List<Long> myRoles = userRoleMapper.selectRoleIdsByUserId(currentUserId);
    List<Long> myExplicit = roleMenuMapper.selectMenuIdsByRoleIds(myRoles);
    Set<Long> mySet = menuService.expandChildren(myExplicit);
    mySet.addAll(myExplicit);

    // 层 3: 授权子集校验
    List<Long> notOwned = menuIds.stream().filter(id -> !mySet.contains(id)).collect(Collectors.toList());
    if (!notOwned.isEmpty()) throw new BusinessException(400, "无权授予未拥有菜单: " + notOwned);

    // 层 4: 内置角色保护
    SysRole target = roleMapper.selectById(targetRoleId);
    boolean isAdmin = userRoleMapper.selectRoleIdsByUserId(currentUserId).stream()
        .map(roleMapper::selectById).anyMatch(r -> "admin".equals(r.getCode()));
    if (target.getIsBuiltin() == 1 && !isAdmin)
        throw new BusinessException(400, "内置角色仅 admin 可改");

    // insert
    LocalDateTime now = LocalDateTime.now();
    for (Long menuId : menuIds) {
        roleMenuMapper.insertOrIgnore(new SysRoleMenu(targetRoleId, menuId, currentUserId, now));
    }

    // 审计 + 事件
    sysLogService.record("GRANT", "role:" + targetRoleId, "menus=" + menuIds);
    eventPublisher.publishEvent(new RoleMenuChangedEvent(this, targetRoleId, currentUserId));
}
```

```javascript
router.beforeEach(async (to, from, next) => {
```
