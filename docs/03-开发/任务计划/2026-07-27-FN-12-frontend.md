# FN-12 前端实施计划（v0.2.0 ③）

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` to implement this plan task-by-task. Steps use checkbox syntax.

**Goal:** 交付 `DictMappingList.vue` 管理页 + `DictMappingParamDialog.vue` 共享参数编辑器 + 3 处向导集成（TransformWizard Step 5 · InterfaceWizard Step 6 · FieldProcess.vue）· 前端 Pinia 全量缓存 + Excel 导入导出 + 权限白名单，让 FN-12 字典功能对用户可见可用。

**Architecture:** RegistryManagement 模式（el-card + toolbar + el-table + dialog）· 三级筛选前端本地 group by · Pinia `useDictMappingStore` 5min TTL 缓存 · 共享 `DictMappingParamDialog.vue` 覆盖 3 处向导集成 · MenuPermission 三角色都加（Q5/A · admin/user 可写 · readonly 只读）。

**Tech Stack:** Vue 3 + Element Plus + Pinia + axios (`@/api/request`) + vitest + vue-draggable-next（**禁 #item slot** · v0.2.0 ③ 不涉及拖拽）

**Spec 源**：[`docs/02-设计/详细设计/2026-07-27-FN-12-frontend-design.md`](../../02-设计/详细设计/2026-07-27-FN-12-frontend-design.md)

## Global Constraints

- 端口固定 5173 · 不变
- 对话与代码注释：中文
- 所有 HTTP 请求走 `@/api/request` axios 实例（响应体已解包为 `{data}` 结构）
- Pinia store 命名规范：`useXxxStore`
- 表单校验用 el-form + `:rules`
- 删除操作必须 el-popconfirm 二次确认
- 一 task 一 commit
- vitest 测试文件放 `frontend/tests/{store,components,views,api}/`
- **禁用 `<template #item>`**（vue-draggable-next v2 不支持 · frontend/CLAUDE.md 明确警告）
- readonly 角色：菜单可见 · 写按钮 `v-if="canWrite"` · 状态 switch `:disabled="!canWrite"`
- 三选一 direction radio 拆条逻辑：`form.mode === 'both'` → 提交 `{direction:1, bidirectional:true}`；否则 `{direction:Number(mode), bidirectional:false}`
- 字段加工参数 dialog 只有二选一 direction（**不含双向** · 单向 lookup）
- `MenuPermission.java` 三角色 ADMIN/USER/READONLY 都追加 `/tools/dict`

- [ ] **Step 1.3: 创建 Pinia store**

`frontend/src/stores/dictMapping.js`：

```js
import { defineStore } from 'pinia'
import { listDictMappings } from '@/api/dictMapping'

const TTL_MS = 5 * 60 * 1000

export const useDictMappingStore = defineStore('dictMapping', {
  state: () => ({
    all: [],
    groupedData: {},   // { systemCode: { dictKey: { direction: [{source, target}] } } }
    refreshedAt: 0
  }),
  getters: {
    systems:    (s) => Object.keys(s.groupedData).sort(),
    dictKeysOf: (s) => (sys) => Object.keys(s.groupedData[sys] || {}).sort()
  },
  actions: {
    async ensureLoaded(force = false) {
      if (!force && Date.now() - this.refreshedAt < TTL_MS && this.all.length > 0) return
      const res = await listDictMappings()
      this.all = res?.data || res || []
      this.groupedData = groupByFn(this.all)
      this.refreshedAt = Date.now()
    },
    invalidate() { this.refreshedAt = 0 }
  }
})

function groupByFn(rows) {
  const g = {}
  for (const r of rows) {
    if (r.status !== 1) continue   // 只装启用条目
    g[r.systemCode]                       ??= {}
    g[r.systemCode][r.dictKey]            ??= {}
    g[r.systemCode][r.dictKey][r.direction] ??= []
    g[r.systemCode][r.dictKey][r.direction].push({ source: r.sourceValue, target: r.targetValue })
  }
  return g
}
```

      { systemCode: 'CIF', dictKey: 'GENDER', direction: 1, sourceValue: 'F', targetValue: '0', status: 1 },

    await s.ensureLoaded()
    expect(s.systems).toEqual([])
  })

  test('invalidate 后 ensureLoaded 会重新拉', async () => {
    listDictMappings.mockResolvedValue({ data: [{ systemCode: 'A', dictKey: 'K', direction: 1, sourceValue: 'x', targetValue: 'y', status: 1 }] })
    const s = useDictMappingStore()

    listDictMappings.mockClear()
    await s.ensureLoaded()  // TTL 内不重复拉
    expect(listDictMappings).not.toHaveBeenCalled()
    s.invalidate()
    await s.ensureLoaded()
    expect(listDictMappings).toHaveBeenCalledTimes(1)
  })

    list.value = res?.data || res || []

    dictKey:     row?.dictKey     || '',
    mode:        row ? String(row.direction) : '1',

    targetValue: row?.targetValue || '',
    cnLabel:     row?.cnLabel     || '',
    status:      row?.status ?? 1
  })

}

async function doSave(skipValidate = false) {
  if (!skipValidate && formRef.value) {
    try { await formRef.value.validate() } catch { return }
  }
  const payload = {
    systemCode: form.systemCode,
    dictKey: form.dictKey,
    sourceValue: form.sourceValue,
    targetValue: form.targetValue,
    cnLabel: form.cnLabel,
    status: form.status,
    bidirectional: form.mode === 'both',
    direction: form.mode === 'both' ? 1 : Number(form.mode)
  }
  if (form.id) await api.updateDictMapping(form.id, payload)
  else         await api.saveDictMapping(payload)
  ElMessage.success('保存成功')
  editVisible.value = false
  await reload()
}

    cnLabel: row.cnLabel, status: row.status,
    direction: row.direction, bidirectional: false
  })
